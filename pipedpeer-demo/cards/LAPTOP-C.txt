######## The day before

# Build the common environment only (slow the first time: about 20 min)
./demo/prep-4-common.sh

# Can this laptop reach LAPTOP-A? Must print "reachable"; if not: SETUP.md, "Can't reach LAPTOP-A"

./demo/0-join.sh <LAPTOP-A IP> <secret>

./demo/2b-cold.sh

./demo/7-gpu.sh

# Step 6 everything (MEM FREE under 3.1 GiB: the second line instead)
./demo/12-everything.sh
