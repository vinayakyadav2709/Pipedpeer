######## The day before. This machine goes first.

# Get the demo folder (no git needed)
curl -L https://github.com/vinayakyadav2709/Pipedpeer/archive/refs/heads/dev.tar.gz | tar xz
cd Pipedpeer-dev/pipedpeer-demo

# Install the nightly and run setup (sudo password when asked); note the sha256 line: every machine must print the same
./demo/prep-0-install.sh

# First machine: make the secret, start the rendezvous as a background service, write join-command.txt
./setup-machine.sh fedora-laptop --rendezvous

# Give the others join-command.txt by pen drive (it holds the secret), or read these two values out
pipedpeer auth show
ip -4 route get 1.1.1.1

# Join yourself (expect "no peers found": normal for the first machine)
pipedpeer join <LAPTOP-A IP>

# Build the demo environments (slow the first time: about 20 min)
./demo/prep-4-common.sh
./demo/prep-3-gpu.sh

# After LAPTOP-B (in a lab: every PC but PC-3) has joined: measure, then check (expect READY)
./demo/prep-2-rehearse.sh
./demo/prep-1-check.sh

######## On the day

# T-30 min: daemon and rendezvous up? (pipedpeer start only if status says not running)
pipedpeer status
pipedpeer start
systemctl --user status pipedpeer-rendezvous
./demo/prep-1-check.sh

# T-10 min: second terminal, keep it up the whole demo
pipedpeer dashboard

# T-10 min: once, as a warm-up
./demo/1-cluster.sh

# Step 1 live join: read the token and your IP out to PC-3
pipedpeer auth show
ip -4 route get 1.1.1.1

# Step 2 the cluster
./demo/1-cluster.sh

# Step 3 headline
./demo/2-headline.sh

# Step 5 GPU: in a spare terminal before it starts; Ctrl+C after step 5
watch -n1 nvidia-smi

# Step 8 failover: at the first "[pipedpeer] remote ok:" line tell PC-3 "stop now"
./demo/9-failover.sh

# Step 9 receipts
./demo/10-proof.sh

# After the demo: stop the rendezvous service
systemctl --user disable --now pipedpeer-rendezvous

# Backup, only if asked: B5 finale, B7 capstone
./demo/13-finale.sh
./demo/14-capstone.sh
